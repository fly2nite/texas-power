name: Daily plan update

on:
  schedule:
    - cron: "0 11 * * *"   # every day at 6 am Texas time
  workflow_dispatch:        # adds a "Run workflow" button

permissions:
  contents: write

jobs:
  crawl:
    runs-on: ubuntu-latest
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Install tools
        run: pip install -r crawler/requirements.txt
      - name: Check the label reader still works
        run: python crawler/tests/test_parser.py
      - name: Find and read every plan
        run: python crawler/run.py
      - name: Save the new plan list
        run: |
          git config user.name "plan-updater"
          git config user.email "actions@users.noreply.github.com"
          git add data/
          git diff --cached --quiet || git commit -m "Plan list for $(date -u +%F)"
          git push
